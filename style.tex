\makeatletter
\setbeamertemplate{frametitle}{%
  \nointerlineskip%
  \@tempdima=\paperwidth
  \advance\@tempdima by -0.28cm
  \hbox{%
    \begin{beamercolorbox}[wd=0.28cm,ht=0.72cm,dp=0.10cm]{frametitle bar}%
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=\@tempdima,ht=0.72cm,dp=0.10cm,%
      leftskip=0.28cm,rightskip=0.20cm]{frametitle}%
      \vbox to 0.72cm{\vfil\usebeamerfont{frametitle}\insertframetitle\vfil}%
    \end{beamercolorbox}%
  }%
}
