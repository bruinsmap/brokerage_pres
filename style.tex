% ===== HEADLINE: section navigation with mini-frame dots =====
% Manual implementation to avoid outer-theme side effects on text width
\newcommand{\navdot}[1]{%
  \ifnum#1=\c@section
    {\color{white}\rule{3pt}{3pt}}%
  \else
    {\color{white!38!navy}\rule{3pt}{3pt}}%
  \fi
}
\setbeamertemplate{headline}{%
  \nointerlineskip%
  \begin{beamercolorbox}[wd=\paperwidth,ht=0.38cm,dp=0.10cm,center]{section in head/foot}%
    \insertsectionnavigationhorizontal{\paperwidth}{\hskip2pt}{\hfill\hskip2pt}%
  \end{beamercolorbox}%
}

% ===== FOOTLINE: short author left + page right =====
\setbeamertemplate{footline}{%
  \leavevmode%

      \hspace{0.35cm}\scriptsize\color{muted}Bruinsma \& Virág%
    \end{beamercolorbox}%
    \begin{beamercolorbox}[wd=0.40\paperwidth,ht=0.26cm,dp=0.14cm]{page number in head/foot}%
      \hfill\scriptsize\color{muted}\insertframenumber\,/\,\inserttotalframenumber\hspace{0.35cm}%
    \end{beamercolorbox}%
  }%
}
