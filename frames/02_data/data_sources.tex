\begin{frame}[fragile]{Patent Data: OECD REGPAT}
    \begin{itemize}
        \item \textbf{Source:} European Patent Office (EPO) PATSTAT database, April 2024 edition
        \item \textbf{Coverage:} All EU-28 countries, 1984--2023 (40 years)
        \item \textbf{Unit of Analysis:} Inventor-window observations in 5-year rolling windows (T1--T8)
        \item \textbf{Geographic Scope:} Inventors assigned to NUTS~2 regions based on registered addresses
    \end{itemize}
