\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{softbox}[title=Reciprocal Brokerage measure]
We proxy brokerage with a constraint-based network position:  \\ 
high constraint means the inventor connects neighborhoods that are not already densely tied to one another.\\[0.08cm]
\textbf{Distribution:} \\ 
mean $=0.203$, median $=0.082$.
\end{softbox}

\column{0.48\textwidth}
\begin{softbox}[title=Atypicality measurement]
Patents are atypical when CPC subclass pairs appear together much less often than expected from historical co-occurrence.\\[0.08cm]
\textbf{Incidence:} 17.5\% of inventor-window observations contain at least one atypical patent at $Z<0$, and 14.6\% do at $Z<-2$.
\end{softbox}
\end{columns}

w_{ij} = \begin{cases} \frac{1}{n_{i}} \sum_{p} w_{ij}^{(p)} & \text{if } j \in \text{Internal} \\ \sum_{p} w_{ij}^{(p)} & \text{if } j \in \text{External} \end{cases}
\]
Weight intrafirm ties inversely by the inventor’s degree within the firm ($n_{i}$). 
\end{eqbox}
