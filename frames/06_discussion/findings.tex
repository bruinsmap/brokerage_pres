\begin{frame}[fragile]{Key Findings}
    \begin{enumerate}
        \item \textbf{Brokerage predicts atypicality} at the individual inventor level
        \begin{itemize}
            \item Provides micro-foundation for regional patterns in literature
        \end{itemize}
        
        \item \textbf{Interfirm brokerage matters more} than intrafirm brokerage
        \begin{itemize}
            \item Organizational boundaries are cognitively meaningful
        \end{itemize}
        
        \item \textbf{Effect is context-dependent} on regional technological diversity
        \begin{itemize}
            \item Broker effectiveness amplified in high-variety regions
        \end{itemize}
    \end{enumerate}
    
    \vspace{0.5cm}
    
    \begin{eqbox}
        \textbf{Overall:} Location in the network matters for innovation, and it matters differently depending on organizational and regional context
    \end{eqbox}
\end{frame}

\begin{frame}[fragile]{Theoretical Implications}
    \begin{itemize}
        \item \textbf{Burt's Structural Holes Theory:} Directly validated at individual + invention level
        \item \textbf{Innovation Economics:} Shows how social structure shapes radical innovation
        \item \textbf{Network Science:} Demonstrates mechanism linking position to outcomes
        \item \textbf{Organizational Boundary Effects:} Firms structure knowledge access and recombination opportunities
    \end{itemize}
    
    \vspace{0.3cm}
    
    \begin{eqbox}
        Individual agency + structural constraint: Brokers are positioned to innovate, but their effectiveness depends on organizational and regional context
    \end{eqbox}
\end{frame}

\begin{frame}[fragile]{Practical Implications}
    \begin{itemize}
        \item \textbf{For Firms:} Diverse collaboration networks predict more radical innovation
        \begin{itemize}
            \item Risk: If only intrafirm brokerage, diversity impact is limited
        \end{itemize}
        
        \item \textbf{For Regions:} Related variety in regional technology base matters
        \begin{itemize}
            \item Regional innovation policy should foster both diversity AND connectivity
        \end{itemize}
        
        \item \textbf{For Researchers:} Individual-level mechanisms matter
        \begin{itemize}
            \item Aggregate patterns are driven by micro-level interactions
        \end{itemize}
    \end{itemize}
\end{frame}
