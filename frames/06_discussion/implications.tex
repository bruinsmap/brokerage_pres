\begin{frame}[fragile]{Limitations}
    \begin{itemize}
        \item \textbf{Patent Data:} Reflects only formally patented innovations (misses non-patented R\&D)
        \item \textbf{Co-inventor Networks:} May not capture all collaboration (e.g., informal consulting)
        \item \textbf{Causality:} Fixed effects and robust specifications help, but endogeneity remains
        \item \textbf{Geography:} Inventor address may not reflect actual workplace
        \item \textbf{Time Windows:} 5-year structure balances dynamics but may miss finer temporal patterns
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Extensions \& Future Research}
    \begin{itemize}
        \item \textbf{Dynamic Models:} Lagged dependent variable to model atypicality persistence
        \item \textbf{Network Formation:} How do inventors select into brokerage positions?
        \item \textbf{Sectoral Variation:} Do mechanisms differ across industries?
        \item \textbf{Firm-Level Aggregation:} How individual brokerage translates to firm innovation
        \item \textbf{International Collaboration:} Border-crossing brokerage vs. domestic
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Policy Recommendations}
    \begin{itemize}
        \item \textbf{Encourage Inter-organizational Collaboration:} More atypical innovation when firms connect
        \item \textbf{Invest in Regional Technological Diversity:} Foster related variety in technology base
        \item \textbf{Network Analysis for Innovation Policy:} Use network metrics to identify future innovators
        \item \textbf{Support Mobility:} Allow inventors to build bridges across organizations and regions
    \end{itemize}
\end{frame}
