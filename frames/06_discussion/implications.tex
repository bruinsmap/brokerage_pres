\begin{frame}{Why this matters beyond one patent dataset}
\begin{columns}[T,onlytextwidth]
\column{0.32\textwidth}
\begin{block}{For innovation theory}
Brokerage is linked to a qualitatively different type of inventive output.
\end{block}

\column{0.32\textwidth}
\begin{block}{For firms}
Firms that want breakthrough recombination should preserve bridging roles across teams and organizations.
\end{block}

\column{0.32\textwidth}
\begin{block}{For places}
Regions gain when they combine related variety with collaborative openness.
\end{block}
\end{columns}

\vspace{0.55cm}

\begin{accentbox}
\textbf{Caveat:} this is patent-based evidence, so informal collaboration and non-patented innovation remain outside the frame.
\end{accentbox}
\end{frame}
