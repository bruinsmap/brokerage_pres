\begin{frame}{Identification strategy}
\scriptsize
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{softbox}[title=Baseline design]
Inventor-level panel regressions link brokerage in window $t$ to the probability that inventor $i$ produces an atypical patent in the same window.\\[0.15cm]
Fixed effects absorb regional differences and common time shocks.
\end{softbox}

\column{0.48\textwidth}
\begin{softbox}[title=Controls and threats]
Controls: degree, patenting intensity, inventor experience.\\[0.15cm]
Main threat: stronger inventors may self-select into brokerage positions.
\end{softbox}
\end{columns}

\vspace{0.25cm}

\begin{eqbox}
\[
\text{Atypical}_{irt}=\beta\,\text{Brokerage}_{it}+\gamma X_{it}+\delta_r+\tau_t+\varepsilon_{irt}
\]
Main outcomes are logit models for binary atypicality thresholds.
\end{eqbox}
\end{frame}
