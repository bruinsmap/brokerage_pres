\begin{frame}[fragile]{Moderating Factors: Organizational Boundaries}
    \textbf{Research Question:} Does the type of structural hole matter?
    
    \vspace{0.3cm}
    
    \begin{itemize}
        \item \textbf{Intrafirm Brokerage:} Bridging clusters within same organization
        \item \textbf{Interfirm Brokerage:} Bridging clusters across different organizations
    \end{itemize}
    
    \vspace{0.3cm}
    
    \begin{eqbox}
        \textbf{Prediction:} Interfirm brokerage should more strongly predict atypicality because:
        \begin{itemize}
            \item Different organizations $\Rightarrow$ distinct knowledge bases
            \item Institutional boundaries create stronger cognitive distance
        \end{itemize}
    \end{eqbox}
\end{frame}

\begin{frame}[fragile]{Moderating Factors: Regional Context}
    \textbf{Regional Technological Relatedness:}
    
    \begin{itemize}
        \item \textbf{Related Variety:} Region has technologies that are related but distinct (good for recombination)
        \item \textbf{Prediction:} In regions with higher related variety, the brokerage effect should be \textit{amplified}
        \item Mechanism: More cognitively accessible raw material for atypical recombination
    \end{itemize}
    
    \vspace{0.3cm}
    
    \begin{eqbox}
        Context matters: Brokerage effectiveness depends on surrounding knowledge environment
    \end{eqbox}
    
    \vspace{0.3cm}
    
    \begin{itemize}
        \item \textbf{Operationalization:} Entropy-based measure of variety in regional technology distribution
    \end{itemize}
\end{frame}
