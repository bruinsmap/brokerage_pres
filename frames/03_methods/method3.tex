Bridging clusters \textbf{within} the same organization
\end{softbox}

\begin{softbox}[title=Interfirm brokerage]
Bridging clusters \textbf{across} different organizations
\end{softbox}
\end{columns}

\vspace{0.25cm}
\begin{eqbox}
\textbf{Prediction:} Interfirm brokerage $\Rightarrow$ \HL{stronger} atypicality effect\\[3pt]
Institutional boundaries create greater cognitive distance
\end{eqbox}
\end{frame}

\begin{frame}{Moderating Factors: Regional Context}

\end{softbox}

\begin{eqbox}
\textbf{Prediction:} Higher related variety \HL{amplifies} the brokerage effect\\[3pt]
Operationalized via entropy-based variety in regional technology distribution
\end{eqbox}
