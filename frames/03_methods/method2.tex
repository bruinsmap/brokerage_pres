\begin{frame}[fragile]{Identification Strategy}
    \textbf{Challenges:}
    \begin{itemize}
        \item Endogeneity: Capable inventors might both secure brokerage AND produce novelty
        \item Selection: Non-random collaboration network formation
        \item Regional Factors: May simultaneously influence network and innovation
    \end{itemize}
    
    \vspace{0.3cm}
    
    \textbf{Solutions:}
    \begin{itemize}
        \item Fixed effects for regions and time periods
        \item Clustered standard errors at regional level
        \item Multiple network definitions (robustness tests)
        \item \textbf{Leverage:} Interfirm vs. intrafirm distinction (mechanism operates more strongly across organizational boundaries)
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Robustness Checks}
    \begin{itemize}
        \item Alternative atypicality thresholds ($Z < -0.5$, $Z < -1.5$, etc.)
        \item Alternative network measures (degree, effective size)
        \item Lag structures (does past brokerage predict current atypicality?)
        \item Sample restrictions (experience, team size, region)
        \item Placebo tests: regionlevel brokerage (should be stronger at individual level)
    \end{itemize}
    
    \vspace{0.5cm}
    
    \begin{eqbox}
        Robustness across specifications strengthens causal inference
    \end{eqbox}
\end{frame}
