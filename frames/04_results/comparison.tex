\begin{frame}[fragile]{Comparison with Alternative Specifications}
    \textbf{Network Measures:}
    \begin{itemize}
        \item Degree (simple connectivity)
        \item Effective size (non-redundant ties)
        \item Constraint (redundancy of neighborhood)
        \item Betweenness (global brokerage)
    \end{itemize}
    
    \vspace{0.3cm}
    
    \begin{eqbox}
        Brokerage-based measures (betweenness, constraint) consistently outpredict simple degree measures
    \end{eqbox}
    
    \vspace{0.3cm}
    
    \textbf{Implication:} It's not just \textit{how many} connections, but \textit{where} they are positioned
\end{frame}

\begin{frame}[fragile]{Time Dynamics}
    \textbf{Do Current or Lagged Network Positions Matter?}
    
    \begin{itemize}
        \item Current-period brokerage $\Rightarrow$ current period atypicality
        \item Prior-period brokerage $\Rightarrow$ current period atypicality
        \item Decay of effect over time
    \end{itemize}
    
    \vspace{0.3cm}
    
    \begin{eqbox}
        Network position effects are relatively immediate (consistent with knowledge from current collaborators)
    \end{eqbox}
\end{frame}
