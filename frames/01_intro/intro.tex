\begin{frame}{Why this question matters}
\small
\begin{columns}[T,onlytextwidth]
\column{0.54\textwidth}
\begin{softbox}[title=The research puzzle]
Most patents are conventional combinations of technologies. We focus on the smaller set that recombines knowledge in statistically unusual ways and ask \textbf{who is structurally positioned to make those combinations happen}.
\end{softbox}

\vspace{0.2cm}
\begin{softbox}[title=Why brokerage should matter]
Inventors who bridge disconnected collaboration neighborhoods see more non-redundant knowledge, more organizational diversity, and more opportunities to connect ideas that are usually kept apart.
\end{softbox}

\column{0.42\textwidth}
\begin{statbox}[title=What is at stake?]
\textbf{Theory:} micro-foundations of novelty\\[0.12cm]
\textbf{Firms:} do cross-boundary brokers help?\\[0.12cm]
\textbf{Regions:} does local variety amplify brokerage?
\end{statbox}

\vspace{0.25cm}
\begin{accentbox}[title=Talk roadmap]
Data $\rightarrow$ mechanism $\rightarrow$ baseline effect $\rightarrow$ robustness $\rightarrow$ interpretation.
\end{accentbox}
\end{columns}

\vspace{0.35cm}
\takeaway{Core intuition: brokers do not just know more people; they stand where distant knowledge pools can actually meet.}
\end{frame}

\begin{frame}{The conference takeaway in one view}
\small
\begin{columns}[T,onlytextwidth]
\column{0.31\textwidth}
\begin{softbox}[title=Question]
Do brokers generate more \textbf{atypical patents} than inventors embedded in denser collaboration neighborhoods?
\end{softbox}

\column{0.31\textwidth}
\begin{softbox}[title=Setting]
EU-28 patenting, 1984--2023, inventor-level panel, time-varying co-inventor networks, CPC-pair rarity as the outcome.
\end{softbox}

\column{0.34\textwidth}
\begin{softbox}[title=Contribution]
We bring the regional brokerage story down to the inventor level and show how organizational boundaries and historical periods reshape the effect.
\end{softbox}
\end{columns}

\vspace{0.35cm}
\takeaway{Working hypothesis: more brokerage should raise the probability of atypical recombination because bridging positions expose inventors to combinations others never encounter.}
\end{frame}
