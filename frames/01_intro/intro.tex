\begin{frame}[fragile]{Introduction}
    \begin{itemize}
        \item Innovation drives economic growth and competitiveness
        \item Most patents are incremental; some are \textbf{breakthrough inventions}
        \item Key question: \textit{Who produces atypical (novel) inventions and under what conditions?}
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Research Question}
    \begin{center}
        {\Large Do inventors in \textbf{brokerage positions} (bridging disconnected clusters in collaboration networks) produce more \textbf{atypical patents}?}
    \end{center}
    
    \vspace{1cm}
    
    \begin{itemize}
        \item Brokerage provides access to diverse, non-redundant knowledge
        \item Atypicality measured via combinatorial rarity of technology classes
        \item Investigation at the \textit{individual inventor} level (micro-foundation of regional patterns)
    \end{itemize}
\end{frame}

\begin{frame}[fragile]{Theoretical Foundation}
    \begin{itemize}
        \item \textbf{Structural Holes (Burt):} Individuals bridging disconnected clusters access more diverse information
        \item \textbf{Atypical Innovation (Uzzi):} Statistical rarity of technology class pairings indicates genuinely novel combinations
        \item \textbf{Co-inventor Networks:} Collaboration maps relational structure and tacit knowledge exchange
    \end{itemize}
    
    \vspace{0.5cm}
    
    \begin{eqbox}
        \textbf{Hypothesis:} Inventors with higher brokerage positions produce more atypical patents
    \end{eqbox}
\end{frame}
