\begin{frame}{Brokerage enrich Innovation }
\footnotesize
\begin{columns}[T,onlytextwidth]
\column{0.54\textwidth}
\begin{softbox}[title=The research puzzle]
Most patents recombine technologies conventionally. We focus on the smaller set that combines knowledge in \textbf{statistically unusual} ways and ask \textbf{who is structurally positioned to make those combinations happen}.
\end{softbox}

\column{0.42\textwidth}
\begin{softbox}[title=What is at stake?]
\textbf{Theory:} micro-foundations of novelty\\[0.06cm]
\textbf{Firms:} do cross-boundary brokers help?\\[0.06cm]
\textbf{Regions:} does local variety amplify brokerage?
\end{softbox}
\end{columns}

\vspace{0.06cm}
\begin{statbox}[title=Why brokerage should matter?]
Inventors bridging \textbf{disconnected collaboration} neighborhoods see more non-redundant knowledge and more opportunities to connect ideas usually kept apart.
\end{statbox}
\end{frame}

\begin{frame}{Theoretical framework}
\small
\begin{columns}[T,onlytextwidth]
\column{0.31\textwidth}
\begin{softbox}[title=Question]
Structural holes theory suggests "brokers" gain access to the diverse information needed for these recombinations. 
\end{softbox}

\column{0.65\textwidth}
\begin{softbox}[title=Question2]

    \item \textbf{Source:} EPO PATSTAT (April 2024).
    \item \textbf{Scope:} EU-25 countries, 1984–2023.
    \item 2.46 million inventor-window observations.
    \item 8 non-overlapping 5-year windows.
\end{softbox}
\end{columns}

\vspace{0.15cm}
\takeaway{More brokerage should raise the probability of atypical recombination because bridging positions expose inventors to combinations others never encounter.}
\end{frame}
